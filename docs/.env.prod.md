## Obrigatórios (fail-closed — `docker compose config -q` falha se faltar)

```env

JEFREY_ENV=prod
JEFREY_DEBUG=false

# API + JWT RS256 (32 bytes = 64 hex chars)
JEFREY_API__SECRET_KEY=<GERE_HEX64>        # ex: python -c "import secrets; print(secrets.token_hex(32))"
JEFREY_API__CORS_ORIGINS=https://seu-dominio.com   # CORS explícito, sem *

# EventBus HMAC (dual-verify kid v1/v2) — CIPHER-033
JEFREY_EVENTBUS__HMAC_KEY=<GERE_HEX64>
JEFREY_EVENTBUS__HMAC_KEYS_JSON={"v1":"<HEX64_V1>","v2":"<HEX64_V2>"}
JEFREY_EVENTBUS__HMAC_KID=v1

# Banco Postgres + pgvector
JEFREY_DATABASE__PASSWORD=<SENHA_FORTE_PG>
JEFREY_DATABASE__USER=jefrey
JEFREY_DATABASE__DB=jefrey

# Redis (working memory)
JEFREY_REDIS__PASSWORD=<SENHA_FORTE_REDIS>

# n8n Basic Auth + API Key
N8N_BASIC_AUTH_USER=jefrey
N8N_BASIC_AUTH_PASSWORD=<SENHA_FORTE_N8N>
N8N_API_KEY=<CHAVE_API_FORTE>

# Grafana Admin
GRAFANA_PASSWORD=<SENHA_FORTE_GRAFANA>
```
